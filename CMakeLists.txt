cmake_minimum_required(VERSION 3.11)
project(CircuitFX C)
add_executable(CircuitFX psp/main.c src/render.c src/wav.c)
target_include_directories(CircuitFX PRIVATE src)
target_compile_options(CircuitFX PRIVATE -O2 -Wall -Wextra)
target_link_libraries(CircuitFX PRIVATE pspdebug pspdisplay pspge pspctrl pspaudio psppower pspuser m)
create_pbp_file(TARGET CircuitFX TITLE "Circuit FX Studio Beta 1" VERSION 00.60)
